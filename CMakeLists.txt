cmake_minimum_required(VERSION 3.16)
project(lazy_lif_dynamic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dynamic_v2 src/dynamic_v2.cpp)
add_executable(holdout_v2 src/holdout_v2.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dynamic_v2 PRIVATE -O3 -DNDEBUG)
  target_compile_options(holdout_v2 PRIVATE -O3 -DNDEBUG)
endif()
